# 第22节：阶段 3 项目实战 — 用 LangGraph + MCP 搭一个多 Agent 系统

## 一、项目概述

把阶段 3 学到的所有框架和协议整合到一个完整项目里：

```
用户提问 → Supervisor 调度 → 
  Researcher（搜索+整理） / Coder（写代码+测试） / Writer（撰写报告）
→ 工具通过 MCP 标准化接入 → Human-in-the-loop 审批关键操作 
→ Checkpointer 持久化状态 → 输出最终结果
```

**项目名**：DevAssistant — 多 Agent 开发助手
**输入**：自然语言的开发任务（如"帮我写一个 Python 爬虫，爬取 Hacker News 首页标题"）
**输出**：可运行的代码 + 技术调研报告 + 代码审查意见

## 三、Step 1 — MCP Server 工具层

先把工具用 MCP 标准化封装，每个工具是一个独立的 MCP Server：

### 3.1 搜索工具 MCP Server

```python
"""
search_server.py — 搜索工具 MCP Server
"""
from mcp.server.fastmcp import FastMCP
import json

mcp = FastMCP("search-tools")

@mcp.tool()
def search_web(query: str, max_results: int = 5) -> str:
    """搜索网络获取相关信息
    
    Args:
        query: 搜索关键词
        max_results: 最多返回结果数（默认5）
    """
    # 生产中调用 SerpAPI / Brave Search API
    # 这里模拟返回
    results = [
        {"title": f"关于 {query} 的文章 {i+1}", "url": f"https://example.com/{i}", "snippet": f"{query} 相关内容..."}
        for i in range(min(max_results, 5))
    ]
    return json.dumps(results, ensure_ascii=False)

@mcp.tool()
def extract_webpage(url: str) -> str:
    """提取网页正文内容
    
    Args:
        url: 网页 URL
    """
    # 生产中用 BeautifulSoup / Trafilatura
    return f"[从 {url} 提取的正文内容...]"

if __name__ == "__main__":
    mcp.run()
```

### 3.2 代码执行 MCP Server

```python
"""
code_server.py — 代码执行工具 MCP Server
"""
from mcp.server.fastmcp import FastMCP
import subprocess
import tempfile
import os

mcp = FastMCP("code-tools")

@mcp.tool()
def execute_python(code: str, timeout: int = 10) -> str:
    """执行 Python 代码并返回结果
    
    Args:
        code: Python 代码字符串
        timeout: 超时秒数（默认10秒）
    """
    with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False) as f:
        f.write(code)
        f.flush()
        try:
            result = subprocess.run(
                ['python', f.name],
                capture_output=True, text=True,
                timeout=timeout
            )
            output = result.stdout if result.returncode == 0 else f"Error:\n{result.stderr}"
            return output[:2000]  # 限制输出长度
        except subprocess.TimeoutExpired:
            return f"执行超时（{timeout}秒）"
        finally:
            os.unlink(f.name)

@mcp.tool()
def run_tests(code: str, test_code: str) -> str:
    """执行代码和对应的测试用例
    
    Args:
        code: 被测试的代码
        test_code: 测试代码（pytest 格式）
    """
    with tempfile.TemporaryDirectory() as tmpdir:
        # 写入源码
        with open(os.path.join(tmpdir, "solution.py"), 'w') as f:
            f.write(code)
        # 写入测试
        with open(os.path.join(tmpdir, "test_solution.py"), 'w') as f:
            f.write(f"from solution import *\n{test_code}")
        # 执行测试
        result = subprocess.run(
            ['python', '-m', 'pytest', '-v', os.path.join(tmpdir, "test_solution.py")],
            capture_output=True, text=True, timeout=30
        )
        return result.stdout + result.stderr

if __name__ == "__main__":
    mcp.run()
```

## 四、Step 2 — 定义 Agent 状态和子图

### 4.1 全局状态

```python
from typing import TypedDict, Annotated
from langgraph.graph import add_messages

class DevAssistantState(TypedDict):
    """主图的全局状态"""
    messages: Annotated[list, add_messages]
    task_type: str                  # supervisor 判断的任务类型
    research_result: str            # 研究子图的输出
    code_result: str                # 编码子图的输出
    review_result: str              # 审查结果
    iteration_count: int            # 迭代计数（防死循环）
    final_output: str               # 最终输出
```

### 4.2 Researcher 子图

```python
from typing import TypedDict
from langgraph.graph import StateGraph, END

class ResearchState(TypedDict):
    query: str
    search_results: list[str]
    analysis: str

def search_node(state: ResearchState) -> dict:
    """搜索相关资料"""
    results = search_web(state["query"])  # 通过 MCP 调用
    return {"search_results": results}

def analyze_node(state: ResearchState) -> dict:
    """分析搜索结果，生成研究报告"""
    research_model = ChatOpenAI(model="gpt-4o")
    response = research_model.invoke([
        SystemMessage(content="你是技术研究专家。根据搜索结果撰写技术调研报告。"),
        HumanMessage(content=f"搜索结果：{state['search_results']}\n\n请撰写调研报告。")
    ])
    return {"analysis": response.content}

# 组装研究子图
research_graph = StateGraph(ResearchState)
research_graph.add_node("search", search_node)
research_graph.add_node("analyze", analyze_node)
research_graph.set_entry_point("search")
research_graph.add_edge("search", "analyze")
research_graph.add_edge("analyze", END)
research_subgraph = research_graph.compile()
```

### 4.3 Coder 子图（带自我修正）

```python
class CoderState(TypedDict):
    task_description: str
    code: str
    test_results: str
    revision_count: int

def generate_code(state: CoderState) -> dict:
    """生成代码"""
    coder_model = ChatOpenAI(model="gpt-4o")
    context = f"任务：{state['task_description']}"
    if state.get("test_results"):
        context += f"\n\n上次测试结果（请修正）：{state['test_results']}"
    
    response = coder_model.invoke([
        SystemMessage(content="""你是 Python 专家。生成高质量、可运行的代码。
        包含错误处理和类型注解。只输出代码，不要解释。"""),
        HumanMessage(content=context)
    ])
    return {"code": response.content, "revision_count": state.get("revision_count", 0) + 1}

def test_code(state: CoderState) -> dict:
    """测试代码"""
    result = execute_python(state["code"])  # 通过 MCP 调用
    return {"test_results": result}

def should_revise(state: CoderState) -> str:
    """判断是否需要修改"""
    if state.get("revision_count", 0) >= 3:
        return END  # 最多修改 3 次
    if "Error" in state.get("test_results", ""):
        return "generate"  # 有错误，重新生成
    return END  # 测试通过

# 组装编码子图（含自我修正循环）
coder_graph = StateGraph(CoderState)
coder_graph.add_node("generate", generate_code)
coder_graph.add_node("test", test_code)
coder_graph.set_entry_point("generate")
coder_graph.add_edge("generate", "test")
coder_graph.add_conditional_edges("test", should_revise)
coder_subgraph = coder_graph.compile()
```

## 五、Step 3 — Supervisor 主图

```python
from langgraph.graph import StateGraph, END
from langgraph.types import interrupt, Command
from langgraph.checkpoint.memory import MemorySaver
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage

supervisor_model = ChatOpenAI(model="gpt-4o")

def supervisor_node(state: DevAssistantState) -> dict:
    """Supervisor：分析任务，决定分配给谁"""
    response = supervisor_model.invoke([
        SystemMessage(content="""你是开发助手的调度中心。根据用户任务和已有结果，决定下一步：
        - "researcher": 需要技术调研（还没有调研结果时）
        - "coder": 需要编写代码（有调研结果或直接编码任务时）
        - "reviewer": 需要代码审查（有代码结果时）
        - "FINISH": 所有工作完成，可以输出最终结果
        只回复一个词。"""),
        *state["messages"],
        HumanMessage(content=f"""
当前状态：
- 调研结果：{"有" if state.get("research_result") else "无"}
- 代码结果：{"有" if state.get("code_result") else "无"}
- 审查结果：{"有" if state.get("review_result") else "无"}
""")
    ])
    return {
        "task_type": response.content.strip().lower(),
        "iteration_count": state.get("iteration_count", 0) + 1
    }

def researcher_wrapper(state: DevAssistantState) -> dict:
    """调用研究子图"""
    query = state["messages"][-1].content if state["messages"] else ""
    result = research_subgraph.invoke({
        "query": query,
        "search_results": [],
        "analysis": ""
    })
    return {
        "research_result": result["analysis"],
        "messages": [AIMessage(content=f"[调研完成]\n{result['analysis']}", name="researcher")]
    }

def coder_wrapper(state: DevAssistantState) -> dict:
    """调用编码子图"""
    task_desc = state["messages"][0].content
    if state.get("research_result"):
        task_desc += f"\n\n参考资料：{state['research_result']}"
    
    result = coder_subgraph.invoke({
        "task_description": task_desc,
        "code": "",
        "test_results": "",
        "revision_count": 0
    })
    return {
        "code_result": result["code"],
        "messages": [AIMessage(content=f"[代码完成]\n```python\n{result['code']}\n```", name="coder")]
    }

def reviewer_node(state: DevAssistantState) -> dict:
    """代码审查（带 Human-in-the-loop）"""
    review_model = ChatOpenAI(model="gpt-4o")
    response = review_model.invoke([
        SystemMessage(content="你是代码审查专家。审查代码的安全性、性能和可维护性。"),
        HumanMessage(content=f"请审查以下代码：\n{state.get('code_result', '')}")
    ])
    
    # 人工确认审查结果
    approval = interrupt({
        "message": "代码审查完成，请确认：",
        "review": response.content,
        "code": state.get("code_result", ""),
        "options": ["approve", "request_changes"]
    })
    
    if approval == "approve":
        return {
            "review_result": response.content,
            "messages": [AIMessage(content=f"[审查通过]\n{response.content}", name="reviewer")]
        }
    else:
        return {
            "code_result": "",  # 清空代码，要求重写
            "messages": [AIMessage(content=f"[审查未通过，需修改]\n{response.content}", name="reviewer")]
        }

def final_output_node(state: DevAssistantState) -> dict:
    """汇总最终输出"""
    output_parts = []
    if state.get("research_result"):
        output_parts.append(f"## 技术调研\n{state['research_result']}")
    if state.get("code_result"):
        output_parts.append(f"## 代码实现\n```python\n{state['code_result']}\n```")
    if state.get("review_result"):
        output_parts.append(f"## 代码审查\n{state['review_result']}")
    
    final = "\n\n---\n\n".join(output_parts)
    return {
        "final_output": final,
        "messages": [AIMessage(content=f"[任务完成]\n{final}")]
    }

# 路由函数
def route_supervisor(state: DevAssistantState) -> str:
    if state.get("iteration_count", 0) >= 10:
        return "final_output"  # 防死循环
    task_type = state.get("task_type", "FINISH")
    if task_type == "researcher":
        return "researcher"
    elif task_type == "coder":
        return "coder"
    elif task_type == "reviewer":
        return "reviewer"
    return "final_output"

# 组装主图
main_graph = StateGraph(DevAssistantState)
main_graph.add_node("supervisor", supervisor_node)
main_graph.add_node("researcher", researcher_wrapper)
main_graph.add_node("coder", coder_wrapper)
main_graph.add_node("reviewer", reviewer_node)
main_graph.add_node("final_output", final_output_node)

main_graph.set_entry_point("supervisor")
main_graph.add_conditional_edges("supervisor", route_supervisor)

for worker in ["researcher", "coder", "reviewer"]:
    main_graph.add_edge(worker, "supervisor")

main_graph.add_edge("final_output", END)

# 编译（带 Checkpointer）
checkpointer = MemorySaver()
app = main_graph.compile(checkpointer=checkpointer)
```

## 六、Step 4 — 运行和调试

```python
def run_dev_assistant(task: str):
    """运行开发助手"""
    config = {"configurable": {"thread_id": f"task-{uuid4()}"}}
    
    print(f"{'='*60}")
    print(f"任务: {task}")
    print('='*60)
    
    try:
        for event in app.stream(
            {"messages": [HumanMessage(content=task)]},
            config=config,
            stream_mode="updates"
        ):
            for node_name, output in event.items():
                print(f"\n--- {node_name} ---")
                for msg in output.get("messages", []):
                    if hasattr(msg, "name"):
                        print(f"  [{msg.name}] {msg.content[:200]}...")
                    else:
                        print(f"  {msg.content[:200]}...")
    except Exception as e:
        # 如果是 interrupt，处理人工审批
        print(f"\n>>> 需要人工操作: {e}")

# 运行
run_dev_assistant("帮我写一个 Python 脚本，爬取 Hacker News 首页的标题列表")
```

## 八、练习题

### 动手编码

1. **基础版**：把上面的代码完整跑通，用模拟数据替代真实 API 调用，验证 Supervisor 调度流程是否正确。

2. **增强版**：
   - 给 Coder 子图添加"生成测试用例"节点——先自动生成 pytest 测试，再用测试结果驱动修正循环
   - 给 Researcher 添加"多轮搜索"能力——第一轮搜索结果不够时自动调整关键词再搜

3. **进阶版**：
   - 把 MCP Server 改成真实的搜索 API（如 Brave Search）和沙箱代码执行
   - 添加 LangSmith Tracing，记录每个节点的输入输出和延迟
   - 部署到服务端，用 PostgresSaver 做持久化

### 面试模拟

4. "讲讲你的多 Agent 项目架构。" 用这个项目作为素材，按 STAR 结构组织：Situation（需求背景）→ Task（要解决什么）→ Action（怎么设计的）→ Result（效果如何）。

5. "你的 Supervisor 做了错误的路由怎么办？" 答：① Worker 自检能力——收到不匹配的任务时拒绝并反馈 Supervisor；② 迭代计数器兜底；③ 事后 Tracing 分析路由日志优化 Supervisor 的 System Prompt。

6. "为什么用 MCP 而不是直接在代码里写工具函数？" 答：① 标准化——换框架不用重写工具；② 隔离——工具运行在独立进程，崩了不影响 Agent；③ 复用——同一个 MCP Server 多个 Agent 共享。
