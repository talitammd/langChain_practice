# 第26节：阶段 4 项目实战 — 搭建一个可追溯来源的企业知识库 Agent

## 一、项目概述

把阶段 4 学到的 RAG 全链路技术整合到一个完整项目里：

```
文档 → 语义切块 → 向量索引 + 知识图谱索引（并行构建）
用户提问 → 查询路由（语义检索 / 图查询）→ 混合检索 + 重排序（或图遍历）
→ 相关性过滤 → 结构化生成（带引用编号）→ 溯源展示
```

**项目名**：EnterpriseKB — 可溯源企业知识库 Agent
**输入**：企业内部文档（制度文件、产品手册、项目复盘）+ 用户自然语言问题
**输出**：基于真实文档的回答，每句话标注引用来源，支持点击查看原文

## 三、Step 1 — 索引构建层

### 3.1 文档处理与双索引构建

```python
from pydantic import BaseModel
from dataclasses import dataclass


@dataclass
class DocumentChunk:
    chunk_id: str
    content: str
    source_doc: str
    metadata: dict


class IndexBuilder:
    def __init__(self, embedding_model, vector_store, graph_store, llm):
        self.embedding_model = embedding_model
        self.vector_store = vector_store
        self.graph_store = graph_store
        self.llm = llm

    def build_index(self, documents: list[dict]) -> None:
        """documents: [{"doc_id": ..., "title": ..., "content": ...}, ...]"""
        for doc in documents:
            chunks = self._semantic_chunk(doc)

            # 向量索引：每块编码后存入向量库
            for chunk in chunks:
                vector = self.embedding_model.encode(chunk.content)
                self.vector_store.upsert(
                    chunk.chunk_id, vector, chunk.content,
                    metadata={"source_doc": chunk.source_doc, **chunk.metadata}
                )

            # 图索引：对每块做实体关系抽取，合并进全局图谱
            for chunk in chunks:
                extraction = self._extract_graph(chunk)
                self.graph_store.merge(extraction, evidence_chunk_id=chunk.chunk_id)

    def _semantic_chunk(self, doc: dict) -> list[DocumentChunk]:
        """沿用第23节的语义切块策略，按标题/段落切分，保留来源元数据"""
        # 简化示意：按段落切分
        paragraphs = [p.strip() for p in doc["content"].split("\n\n") if p.strip()]
        return [
            DocumentChunk(
                chunk_id=f"{doc['doc_id']}_chunk_{i}",
                content=p,
                source_doc=doc["title"],
                metadata={"doc_id": doc["doc_id"], "chunk_index": i},
            )
            for i, p in enumerate(paragraphs)
        ]

    def _extract_graph(self, chunk: DocumentChunk):
        """沿用第25节的实体关系抽取"""
        prompt = f"从以下文本中抽取实体和关系，输出结构化三元组：\n{chunk.content}"
        return self.llm.with_structured_output(ExtractionResult).invoke(prompt)
```

## 四、Step 2 — 查询路由层

```python
from enum import Enum

class QueryRoute(str, Enum):
    SEMANTIC = "semantic"
    RELATIONAL = "relational"


def route_query(llm, question: str) -> QueryRoute:
    """判断查询走语义检索还是图查询"""
    prompt = f"""判断以下问题的类型：
- 如果是查找具体信息、解释说明类问题，返回 semantic
- 如果是查询实体之间关系、需要综合多方信息的问题，返回 relational

问题：{question}
只返回 semantic 或 relational。"""
    response = llm.invoke(prompt)
    return QueryRoute(response.content.strip().lower())
```

## 五、Step 3 — 混合检索 + 重排序（语义查询路径）

```python
class SemanticRetrievalPipeline:
    def __init__(self, vector_store, bm25_index, reranker, relevance_threshold: float = 0.3):
        self.vector_store = vector_store
        self.bm25_index = bm25_index
        self.reranker = reranker
        self.relevance_threshold = relevance_threshold

    def retrieve(self, query: str) -> list[dict]:
        vector_results = self.vector_store.search(query, top_k=20)
        bm25_results = self.bm25_index.search(query, top_k=20)
        merged = self._rrf_merge(vector_results, bm25_results)
        reranked = self.reranker.rerank(query, merged, top_k=5)
        # 相关性阈值过滤：宁可承认"不知道"，也不能让模型硬凑答案
        return [doc for doc in reranked if doc["score"] > self.relevance_threshold]

    def _rrf_merge(self, list_a: list, list_b: list, k: int = 60) -> list[dict]:
        scores, docs = {}, {}
        for rank, doc in enumerate(list_a):
            scores[doc["id"]] = scores.get(doc["id"], 0) + 1 / (k + rank + 1)
            docs[doc["id"]] = doc
        for rank, doc in enumerate(list_b):
            scores[doc["id"]] = scores.get(doc["id"], 0) + 1 / (k + rank + 1)
            docs[doc["id"]] = doc
        return [docs[i] for i in sorted(scores, key=lambda i: -scores[i])]
```

## 六、Step 4 — 图查询路径（关系查询）

```python
class RelationalRetrievalPipeline:
    def __init__(self, graph_store, llm):
        self.graph_store = graph_store
        self.llm = llm

    def retrieve(self, question: str) -> list[dict]:
        main_entity = self._extract_entity(question)
        subgraph = self.graph_store.get_neighborhood(main_entity, max_hops=2)
        return [
            {
                "content": f"{r.source} --{r.relation_type}--> {r.target}",
                "source_doc": r.evidence_chunk_id,
                "score": 1.0,  # 图查询是确定性遍历，不需要相似度打分
            }
            for r in subgraph.relations
        ]

    def _extract_entity(self, question: str) -> str:
        response = self.llm.invoke(f"从这个问题中提取核心实体名称，只返回名称：{question}")
        return response.content.strip()
```

## 七、Step 5 — 结构化生成与溯源

带引用编号的结构化输出，是溯源能力的核心：

```python
from pydantic import BaseModel, Field

class Citation(BaseModel):
    claim: str = Field(..., description="回答中的一个具体论断")
    evidence_ids: list[str] = Field(..., description="支撑这个论断的证据编号列表")

class TraceableAnswer(BaseModel):
    answer: str = Field(..., description="完整的回答文本")
    citations: list[Citation] = Field(..., description="每个论断对应的引用")
    confidence: str = Field(..., description="high/medium/low，模型对答案可靠性的自评")


class AnswerGenerator:
    def __init__(self, llm):
        self.llm = llm

    def generate(self, question: str, evidence: list[dict]) -> TraceableAnswer:
        evidence_text = "\n".join(
            f"[证据{i+1}] {doc['content']}（来源：{doc['source_doc']}）"
            for i, doc in enumerate(evidence)
        )
        prompt = f"""请仅根据以下证据回答问题。回答中的每个具体论断都要标注引用了哪几号证据。
如果证据不足以回答问题，请在 answer 中明确说明，并将 confidence 设为 low。

证据：
{evidence_text}

问题：{question}"""

        return self.llm.with_structured_output(TraceableAnswer).invoke(prompt)


def resolve_citations_to_sources(answer: TraceableAnswer, evidence: list[dict]) -> dict:
    """把引用编号映射回真实的文档来源，供前端展示"""
    trace_map = {}
    for citation in answer.citations:
        sources = []
        for eid in citation.evidence_ids:
            idx = int(eid.replace("证据", "")) - 1
            if 0 <= idx < len(evidence):
                sources.append({
                    "source_doc": evidence[idx]["source_doc"],
                    "content": evidence[idx]["content"],
                })
        trace_map[citation.claim] = sources
    return trace_map
```

## 八、Step 6 — 整体串联

```python
class EnterpriseKBAgent:
    def __init__(self, llm, semantic_pipeline, relational_pipeline, generator):
        self.llm = llm
        self.semantic_pipeline = semantic_pipeline
        self.relational_pipeline = relational_pipeline
        self.generator = generator

    def ask(self, question: str) -> dict:
        route = route_query(self.llm, question)

        if route == QueryRoute.SEMANTIC:
            evidence = self.semantic_pipeline.retrieve(question)
        else:
            evidence = self.relational_pipeline.retrieve(question)

        if not evidence:
            return {
                "answer": "抱歉，知识库中没有找到与您问题相关的信息。",
                "trace": {},
                "route": route.value,
            }

        answer = self.generator.generate(question, evidence)
        trace = resolve_citations_to_sources(answer, evidence)

        return {
            "answer": answer.answer,
            "confidence": answer.confidence,
            "trace": trace,
            "route": route.value,
        }


# 使用示例
agent = EnterpriseKBAgent(llm, semantic_pipeline, relational_pipeline, generator)
result = agent.ask("公司报销制度里，差旅费的审批流程是什么？")
print(result["answer"])
for claim, sources in result["trace"].items():
    print(f"\n论断：{claim}")
    for s in sources:
        print(f"  来源：{s['source_doc']}")
```

## 十、练习题

### 动手编码

1. **基础版**：把上面的代码用模拟数据跑通，验证路由判断能否正确区分"公司报销流程是什么"（语义查询）和"哪些部门和财务部有审批关系"（关系查询）。

2. **增强版**：
   - 给 `SemanticRetrievalPipeline` 加入 Matryoshka 向量的两阶段检索（低维粗筛+高维精排）
   - 给 `RelationalRetrievalPipeline` 加入全局查询能力（社区摘要检索）
   - 加一个"证据去重"环节，避免同一份文档的相邻块被重复引用

3. **进阶版**：
   - 把向量库换成真实的 Qdrant/Milvus，图存储换成 Neo4j
   - 加入增量索引更新机制（复用第24节的 `IncrementalIndexer`）
   - 用第27节要讲的 Eval 框架，对这个系统的检索准确率和回答质量做量化评估

### 面试模拟

4. "为什么你的知识库 Agent 要同时维护向量索引和图索引？" 参考答法：两者解决不同类型的问题，向量索引擅长语义相似度匹配，图索引擅长关系推理和多跳查询，同时维护、按需路由，比强行用一种索引覆盖所有场景更精确也更省成本。

5. "如果模型引用的证据编号是错的（比如引用了不存在的证据5），怎么防御？" 参考答法：在 `resolve_citations_to_sources` 里对引用编号做边界检查，超出范围的引用直接丢弃并记录日志用于后续分析模型的引用准确率，不能让前端展示出错误的溯源信息。

6. "这个系统怎么防止用户问一些知识库之外、甚至有害的问题？" 参考答法：相关性阈值过滤是第一道防线——检索不到相关证据就直接拒答；同时可以在生成层加一层内容安全检查，对涉及企业敏感信息之外的请求（如寻求医疗法律建议）做范围限定回复。

## 十一、小结

这个项目把阶段 4 的四节课串成了一条完整链路：语义切块和混合检索打好检索地基，向量数据库和 Embedding 选型决定检索效率，GraphRAG 补齐多跳关系推理的短板，查询路由让系统按需选择最合适的检索路径，结构化输出的引用机制让每一句回答都有据可查。这是一个信息密度很高的面试项目素材，能从"为什么不是简单的向量检索"一路讲到路由设计和溯源机制。
