## 二、项目架构总览

整个项目只有三个核心组件：

| 组件 | 职责 | 对应前面哪节 |
|---|---|---|
| **工具注册表** | 字典：函数名 → Python 函数 + schema | 第05节 Function Calling |
| **对话记忆** | 列表：存所有 messages（用户/模型/工具） | 第03节 Prompt + 第04节 结构化输出 |
| **Agent Loop** | while 循环：调模型→有 tool_calls 就执行并回填→没有就返回 | 第05/06节 |

核心流程图（文字版）：

```
用户输入 → [messages 追加 user]
          → while True:
              调模型(messages, tools)
              if 有 tool_calls:
                  执行工具 → 结果追加到 messages
              else:
                  返回模型文本 → 结束
              if 超过最大轮次:
                  强制退出（安全阀）
```

## 三、完整项目代码

### 3.1 主文件：mini_agent.py

```python
"""
极简 Agent —— 不依赖框架，~90 行实现完整工具循环
依赖：pip install openai pydantic jinja2
"""
import json
from typing import Any, Callable
from pydantic import BaseModel
from jinja2 import Template
from openai import OpenAI

# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 1. Prompt 模板（Jinja2 拆三块）
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
SYSTEM_TEMPLATE = Template("""
你是一个{{ role }}。
约束：
- 回答语言：{{ lang }}
- 如果不确定答案，诚实说"我不知道"
- 尽量简洁
""".strip())

def build_system_prompt(role: str = "通用助手", lang: str = "中文") -> str:
    return SYSTEM_TEMPLATE.render(role=role, lang=lang)


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 2. 工具注册表
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
class ToolRegistry:
    """工具白名单：注册函数 + schema，Agent 只能调注册过的"""
    def __init__(self):
        self._tools: dict[str, Callable] = {}
        self._schemas: list[dict] = []

    def register(self, name: str, description: str, parameters: dict, func: Callable):
        self._tools[name] = func
        self._schemas.append({
            "type": "function",
            "function": {"name": name, "description": description, "parameters": parameters},
        })

    @property
    def schemas(self) -> list[dict]:
        return self._schemas

    def execute(self, name: str, arguments: dict) -> str:
        if name not in self._tools:
            return f"[错误] 未注册的工具: {name}"
        try:
            result = self._tools[name](**arguments)
            return json.dumps(result, ensure_ascii=False) if not isinstance(result, str) else result
        except Exception as e:
            return f"[工具执行失败] {type(e).__name__}: {e}"


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 3. Agent Loop（核心）
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
class MiniAgent:
    def __init__(self, registry: ToolRegistry, model: str = "gpt-4o", max_turns: int = 10):
        self.client = OpenAI()
        self.registry = registry
        self.model = model
        self.max_turns = max_turns
        self.messages: list[dict] = [
            {"role": "system", "content": build_system_prompt()},
        ]

    def chat(self, user_input: str) -> str:
        """Agent 主入口：发问 → 循环调工具 → 返回最终文本"""
        self.messages.append({"role": "user", "content": user_input})

        for turn in range(self.max_turns):
            resp = self.client.chat.completions.create(
                model=self.model,
                messages=self.messages,
                tools=self.registry.schemas or None,
            )
            msg = resp.choices[0].message
            self.messages.append(msg.model_dump(exclude_none=True))

            # —— 没有 tool_calls → 模型觉得够了，返回文本 ——
            if not msg.tool_calls:
                return msg.content or ""

            # —— 有 tool_calls → 执行工具，结果回填 ——
            for call in msg.tool_calls:
                args = json.loads(call.function.arguments)
                result = self.registry.execute(call.function.name, args)
                self.messages.append({
                    "role": "tool",
                    "tool_call_id": call.id,
                    "content": result,
                })

        return "[Agent] 达到最大轮次限制，强制退出"


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 4. 注册工具 & 运行
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
def get_weather(city: str, unit: str = "c") -> str:
    """模拟天气 API（真实场景替换为 requests.get(...)）"""
    fake_data = {"北京": 28, "上海": 31, "广州": 33}
    temp = fake_data.get(city, 25)
    return f"{city} 晴 {temp}°{'C' if unit == 'c' else 'F'}"

def calculate(expression: str) -> str:
    """安全计算器（仅允许数字和基本运算符）"""
    allowed = set("0123456789+-*/.() ")
    if not all(ch in allowed for ch in expression):
        return "[错误] 含非法字符，拒绝执行"
    try:
        return str(eval(expression))   # 生产里用 ast.literal_eval 或 sympy
    except Exception as e:
        return f"[计算失败] {e}"


if __name__ == "__main__":
    reg = ToolRegistry()
    reg.register(
        name="get_weather",
        description="查询指定城市的实时天气",
        parameters={
            "type": "object",
            "properties": {
                "city": {"type": "string", "description": "城市名"},
                "unit": {"type": "string", "enum": ["c", "f"], "description": "温度单位"},
            },
            "required": ["city"],
        },
        func=get_weather,
    )
    reg.register(
        name="calculate",
        description="计算数学表达式",
        parameters={
            "type": "object",
            "properties": {
                "expression": {"type": "string", "description": "数学表达式，如 3*5+2"},
            },
            "required": ["expression"],
        },
        func=calculate,
    )

    agent = MiniAgent(reg)
    # 多轮对话演示
    print(agent.chat("北京和上海今天天气怎么样？"))
    print(agent.chat("那两个城市的温度差是多少？"))
```

**逐块对应表**：

| 代码块 | 用了哪节知识 |
|---|---|
| `SYSTEM_TEMPLATE` + Jinja2 | 第03节 Prompt 模板化 |
| `ToolRegistry.schemas` | 第05节 Function Calling 的 JSON Schema 定义 |
| `ToolRegistry.execute` + 白名单 | 第05节安全护栏 |
| `msg.tool_calls` 判断 + 循环 | 第05节三轮闭环 → 放进 while loop |
| `max_turns` 兜底 | 生产安全阀 |
| `self.messages` 列表 | 对话记忆 |

### 3.2 扩展：加流式输出版本

```python
def chat_stream(self, user_input: str):
    """流式版 Agent Loop：最后一轮用 stream=True"""
    self.messages.append({"role": "user", "content": user_input})

    for turn in range(self.max_turns):
        # 工具循环阶段用非流式（需要完整 tool_calls）
        resp = self.client.chat.completions.create(
            model=self.model,
            messages=self.messages,
            tools=self.registry.schemas or None,
        )
        msg = resp.choices[0].message
        self.messages.append(msg.model_dump(exclude_none=True))

        if not msg.tool_calls:
            # 最后一轮：重新请求用流式输出给用户
            self.messages.pop()  # 移除刚才的非流式回复
            stream = self.client.chat.completions.create(
                model=self.model,
                messages=self.messages,
                stream=True,
            )
            full_text = ""
            for chunk in stream:
                content = chunk.choices[0].delta.content or ""
                print(content, end="", flush=True)
                full_text += content
            print()
            self.messages.append({"role": "assistant", "content": full_text})
            return full_text

        for call in msg.tool_calls:
            args = json.loads(call.function.arguments)
            result = self.registry.execute(call.function.name, args)
            self.messages.append({"role": "tool", "tool_call_id": call.id, "content": result})

    return "[Agent] 达到最大轮次限制"
```

**工程要点**：工具循环阶段需要完整的 `tool_calls`，所以用非流式；最终回复阶段切到流式给用户逐字推送。这是生产 Agent 的常见做法。

## 七、练习题参考答案

<details>
<summary>点击展开 A 类参考要点</summary>

**A1.** 正常退出条件：模型返回的 message 没有 `tool_calls`，说明它判断不需要再调工具、可以直接回答了。`max_turns` 兜底是防止模型陷入死循环（反复调工具不停下来），线上必须有这个安全阀，否则一个请求可能烧掉大量 token 和时间。

**A2.** ①白名单机制——只有注册过的工具才能被调用，模型输出未注册的函数名直接拒绝；②执行层用 try/except 包裹，单个工具失败不会崩整个 Agent；③可以扩展：加参数校验（Pydantic）、危险操作人工确认、审计日志。

**A3.** 工具循环阶段需要拿到完整的 `tool_calls`（函数名+参数），流式时 tool_calls 以 delta 分片到达，要攒完才能执行——用非流式更简单可靠。最终回复阶段是纯文本输出给用户，这时用流式逐字推送体验更好。理论上全程流式也行，但需要额外处理攒分片逻辑，复杂度高。

</details>

<details>
<summary>点击展开 B 类参考思路</summary>

**B1.** 在 `ToolRegistry` 里加 `def unregister(self, name): self._tools.pop(name, None); self._schemas = [s for s in self._schemas if s["function"]["name"] != name]`。

**B4.** `def export_history(self) -> list[dict]: return [m if isinstance(m, dict) else m.model_dump(exclude_none=True) for m in self.messages]`。可以 `json.dumps(agent.export_history(), ensure_ascii=False, indent=2)` 落文件。
