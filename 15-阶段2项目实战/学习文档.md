# 第15节：阶段 2 项目实战 — 智能研究助手 Agent

## 一、项目概述

把阶段 2 学到的所有范式整合到一个能跑的 Agent 里：

```
用户提问 → Plan-and-Execute 拆任务 → ReAct 循环执行 → 
工具失败重试降级 → 上下文工程管理 token → 记忆系统跨任务共享 → 
Reflexion 反思修正 → 输出报告
```

**项目名**：ResearchAgent — 智能研究助手
**输入**：自然语言研究问题（如"对比 React 和 Vue 在 2026 年的性能表现"）
**输出**：结构化研究报告（含数据来源）

### 3.2 错误处理器（第12节）

```python
import random

class CircuitBreaker:
    """熔断器：连续失败 N 次后跳过该工具"""
    def __init__(self, failure_threshold: int = 3, recovery_time: int = 60):
        self.failure_threshold = failure_threshold
        self.recovery_time = recovery_time
        self.failures: dict[str, int] = {}  # tool_name -> failure_count
        self.opened_at: dict[str, float] = {}  # tool_name -> timestamp

    def can_call(self, tool_name: str) -> bool:
        if tool_name not in self.failures or self.failures[tool_name] < self.failure_threshold:
            return True
        # 检查是否过了恢复时间
        if tool_name in self.opened_at:
            if time.time() - self.opened_at[tool_name] > self.recovery_time:
                self.failures[tool_name] = 0
                del self.opened_at[tool_name]
                return True
        return False

    def record_failure(self, tool_name: str):
        self.failures[tool_name] = self.failures.get(tool_name, 0) + 1
        if self.failures[tool_name] >= self.failure_threshold:
            self.opened_at[tool_name] = time.time()

    def record_success(self, tool_name: str):
        self.failures[tool_name] = 0


def retry_with_backoff(func, max_retries=3, base_delay=1.0):
    """指数退避重试"""
    for attempt in range(max_retries):
        result = func()
        if result.success:
            return result
        # 瞬时错误才重试（超时、限流），持久错误不重试
        if not is_transient_error(result.error):
            return result
        delay = base_delay * (2 ** attempt) + random.uniform(0, 0.5)
        time.sleep(delay)
    return result  # 返回最后一次失败结果


def is_transient_error(error: str) -> bool:
    """判断是否为可重试的瞬时错误"""
    transient_keywords = ["超时", "timeout", "限流", "rate limit", "503", "502"]
    return any(kw in error.lower() for kw in transient_keywords)


class ToolExecutor:
    """工具执行器：集成重试 + 熔断 + 降级"""
    
    FALLBACK = {
        "search": "extract",  # 搜索失败时尝试直接提取已知URL
    }

    def __init__(self):
        self.breaker = CircuitBreaker()

    def execute(self, tool_name: str, **kwargs) -> ToolResult:
        # 熔断检查
        if not self.breaker.can_call(tool_name):
            print(f"[CircuitBreaker] {tool_name} 已熔断，尝试降级")
            return self._fallback(tool_name, **kwargs)

        # 重试执行
        result = retry_with_backoff(lambda: TOOLS[tool_name](**kwargs))

        if result.success:
            self.breaker.record_success(tool_name)
        else:
            self.breaker.record_failure(tool_name)
            if not self.breaker.can_call(tool_name):
                return self._fallback(tool_name, **kwargs)

        return result

    def _fallback(self, tool_name: str, **kwargs) -> ToolResult:
        fallback_name = self.FALLBACK.get(tool_name)
        if fallback_name and self.breaker.can_call(fallback_name):
            print(f"[Fallback] {tool_name} → {fallback_name}")
            return TOOLS[fallback_name](**kwargs)
        return ToolResult(False, None, f"{tool_name} 不可用且无降级方案")
```

### 3.3 记忆系统（第11节）

```python
class MemoryStore:
    """跨子任务的记忆系统"""
    
    def __init__(self):
        self.long_term: list[dict] = []  # 长期记忆（向量库模拟）
        self.episodic: list[str] = []    # 情景记忆（Reflexion 反思）

    def write(self, content: str, metadata: dict = None):
        """写入长期记忆"""
        self.long_term.append({
            "content": content,
            "metadata": metadata or {},
            "timestamp": time.time(),
        })

    def retrieve(self, query: str, top_k: int = 3) -> list[str]:
        """检索相关记忆（生产中用向量相似度搜索）"""
        # 简化版：关键词匹配
        scored = []
        for item in self.long_term:
            score = sum(1 for word in query.split() if word in item["content"])
            scored.append((score, item["content"]))
        scored.sort(key=lambda x: -x[0])
        return [content for score, content in scored[:top_k] if score > 0]

    def add_reflection(self, reflection: str):
        """添加 Reflexion 反思到情景记忆"""
        self.episodic.append(reflection)

    def get_reflections(self) -> str:
        """获取所有反思（注入下一轮 prompt）"""
        if not self.episodic:
            return ""
        return "\n".join([f"过去的经验教训:"]
                        + [f"- {r}" for r in self.episodic])
```

### 3.4 上下文管理器（第13节）

```python
class ContextManager:
    """Token 预算管理 + 上下文压缩"""
    
    def __init__(self, total_budget: int = 128000):
        self.budget = {
            "system": int(total_budget * 0.08),
            "history": int(total_budget * 0.23),
            "retrieval": int(total_budget * 0.16),
            "tool_results": int(total_budget * 0.16),
            "generation": int(total_budget * 0.37),
        }
        self.usage = {k: 0 for k in self.budget}
        self.external_store: dict[str, str] = {}  # Write 的外部存储

    def estimate_tokens(self, text: str) -> int:
        """粗略估算 token 数（中文约1.5字/token，英文约4字符/token）"""
        return len(text) // 3  # 简化估算

    def write_external(self, key: str, content: str):
        """Write 操作：存到外部，释放上下文"""
        self.external_store[key] = content

    def read_external(self, key: str) -> str:
        """从外部存储读取"""
        return self.external_store.get(key, "")

    def select(self, query: str, memory: MemoryStore, max_tokens: int = None) -> str:
        """Select 操作：检索记忆注入上下文"""
        max_tokens = max_tokens or self.budget["retrieval"]
        results = memory.retrieve(query, top_k=5)
        
        selected = []
        total = 0
        for r in results:
            tokens = self.estimate_tokens(r)
            if total + tokens > max_tokens:
                break
            selected.append(r)
            total += tokens
        return "\n".join(selected)

    def compress(self, text: str, target_ratio: float = 0.3) -> str:
        """Compress 操作：压缩冗长文本"""
        if self.estimate_tokens(text) < 500:
            return text  # 太短不需要压缩
        # 生产中: 调用 LLM 压缩
        response = openai.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{
                "role": "user",
                "content": f"压缩以下内容，保留关键数据和结论，目标长度为原文的{target_ratio*100}%:\n\n{text}"
            }],
        )
        return response.choices[0].message.content

    def can_fit(self, category: str, text: str) -> bool:
        tokens = self.estimate_tokens(text)
        return self.usage[category] + tokens <= self.budget[category]

    def allocate(self, category: str, text: str) -> bool:
        if not self.can_fit(category, text):
            return False
        self.usage[category] += self.estimate_tokens(text)
        return True
```

### 3.5 ReAct 执行器（第9节 + 第10节）

```python
class ReActExecutor:
    """单个子任务的 ReAct 执行器"""
    
    def __init__(self, tool_executor: ToolExecutor, context: ContextManager, 
                 memory: MemoryStore):
        self.tool_executor = tool_executor
        self.context = context
        self.memory = memory
        self.max_steps = 10

    def execute(self, subtask: str) -> str:
        """执行单个子任务，返回结果"""
        # 注入相关记忆
        memory_context = self.context.select(subtask, self.memory)
        reflections = self.memory.get_reflections()

        messages = [
            {"role": "system", "content": f"""你是一个研究助手。使用 ReAct 模式完成任务。

可用工具: search(query), extract(url), code(script)

每一步输出格式:
Thought: 你的思考
Action: tool_name(参数)
"""
            + (f"\n\n相关记忆:\n{memory_context}" if memory_context else "")
            + (f"\n\n{reflections}" if reflections else "")
            },
            {"role": "user", "content": f"任务: {subtask}"}
        ]

        scratchpad = ""  # 记录 Thought-Action-Observation 链

        for step in range(self.max_steps):
            # 生成 Thought + Action
            response = openai.chat.completions.create(
                model="gpt-4o",
                messages=messages + [{"role": "assistant", "content": scratchpad}],
            )
            step_output = response.choices[0].message.content
            scratchpad += step_output + "\n"

            # 检查是否完成
            if "Final Answer:" in step_output or "最终答案:" in step_output:
                final = step_output.split("Final Answer:")[-1].split("最终答案:")[-1].strip()
                # 把中间过程 Write 到外部（Isolate 思想）
                self.context.write_external(f"scratchpad_{subtask[:20]}", scratchpad)
                return final

            # 解析并执行 Action
            action_result = self._parse_and_execute(step_output)
            scratchpad += f"Observation: {action_result}\n"

            # 上下文管理：scratchpad 太长就压缩
            if self.context.estimate_tokens(scratchpad) > 4000:
                scratchpad = self.context.compress(scratchpad, target_ratio=0.3)
                print(f"[Compress] scratchpad 压缩到 30%")

        return "达到最大步数，未能完成任务"

    def _parse_and_execute(self, output: str) -> str:
        """解析 Action 并执行工具"""
        # 简化解析
        if "search(" in output:
            query = output.split("search(")[1].split(")")[0].strip("'\"")
            result = self.tool_executor.execute("search", query=query)
            return self._translate_result(result, "search")
        elif "extract(" in output:
            url = output.split("extract(")[1].split(")")[0].strip("'\"")
            result = self.tool_executor.execute("extract", url=url)
            return self._translate_result(result, "extract")
        elif "code(" in output:
            code = output.split("code(")[1].rsplit(")", 1)[0].strip("'\"")
            result = self.tool_executor.execute("code", code=code)
            return self._translate_result(result, "code")
        return "未识别的 Action"

    def _translate_result(self, result: ToolResult, tool_name: str) -> str:
        """错误信息翻译：技术报错 → Agent 可理解的提示"""
        if result.success:
            # 成功结果如果太长，做压缩
            data_str = str(result.data)
            if self.context.estimate_tokens(data_str) > 2000:
                data_str = self.context.compress(data_str, target_ratio=0.3)
            return data_str
        else:
            # 翻译错误信息
            error_map = {
                "timeout": "搜索超时，请换一个关键词重试",
                "超时": "搜索超时，请换一个关键词重试",
                "rate limit": "搜索服务繁忙，建议使用其他工具",
                "限流": "搜索服务繁忙，建议使用其他工具",
            }
            for key, friendly in error_map.items():
                if key in result.error.lower():
                    return f"工具执行失败: {friendly}"
            return f"工具执行失败: {result.error}"
```

### 3.6 Reflexion 报告审查（第14节）

```python
class ReflexionChecker:
    """对最终报告做 Reflexion 审查"""
    
    def __init__(self, max_attempts: int = 2):
        self.max_attempts = max_attempts

    def check_and_refine(self, report: str, task: str) -> str:
        reflections = []

        for attempt in range(self.max_attempts):
            # 评估报告质量
            critique = self._critique(report, task)

            if "无需修改" in critique:
                print(f"[Reflexion] 第{attempt+1}次审查通过")
                return report

            # 生成反思
            reflection = self._reflect(report, critique)
            reflections.append(reflection)
            print(f"[Reflexion] 第{attempt+1}次审查发现问题: {reflection[:80]}...")

            # 带着反思修改
            report = self._refine(report, critique, "\n".join(reflections))

        return report

    def _critique(self, report: str, task: str) -> str:
        """审查报告"""
        response = openai.chat.completions.create(
            model="gpt-4o",
            messages=[{
                "role": "user",
                "content": f"""审查以下研究报告:

任务: {task}
报告: {report}

检查:
1. 数据是否有来源支撑？
2. 结论是否有逻辑跳跃？
3. 是否遗漏了重要方面？

如果没有问题就说"无需修改"。"""
            }],
        )
        return response.choices[0].message.content

    def _reflect(self, report: str, critique: str) -> str:
        """生成具体反思"""
        response = openai.chat.completions.create(
            model="gpt-4o",
            messages=[{
                "role": "user",
                "content": f"报告问题: {critique}\n\n请用一句话总结经验教训。"
            }],
        )
        return response.choices[0].message.content

    def _refine(self, report: str, critique: str, reflections: str) -> str:
        """根据反思修改报告"""
        response = openai.chat.completions.create(
            model="gpt-4o",
            messages=[{
                "role": "user",
                "content": f"""根据以下反馈修改报告:

原报告: {report}
审查反馈: {critique}
历史经验: {reflections}

请修改并输出完整报告。"""
            }],
        )
        return response.choices[0].message.content
```

### 3.7 完整 Agent 整合

```python
class ResearchAgent:
    """智能研究助手 — 整合阶段2所有范式"""
    
    def __init__(self):
        self.tool_executor = ToolExecutor()
        self.memory = MemoryStore()
        self.context = ContextManager(total_budget=128000)
        self.reflexion = ReflexionChecker(max_attempts=2)
        self.executor = ReActExecutor(self.tool_executor, self.context, self.memory)

    def run(self, question: str) -> str:
        print(f"{'='*60}")
        print(f"研究问题: {question}")
        print(f"{'='*60}")

        # === 第一步：Plan-and-Execute 拆任务 ===
        subtasks = self._plan(question)
        print(f"\n[Plan] 拆分为 {len(subtasks)} 个子任务:")
        for i, t in enumerate(subtasks):
            print(f"  {i+1}. {t}")

        # === 第二步：逐个执行子任务 ===
        results = []
        for i, subtask in enumerate(subtasks):
            print(f"\n[Execute] 子任务 {i+1}/{len(subtasks)}: {subtask}")
            
            # 每个子任务用独立上下文（Isolate）
            result = self.executor.execute(subtask)
            results.append({"subtask": subtask, "result": result})
            
            # 关键发现写入长期记忆（Write）
            self.memory.write(result, metadata={"subtask": subtask})
            
            # 压缩结果存外部，不占主上下文（Compress + Write）
            compressed = self.context.compress(result, target_ratio=0.2)
            self.context.write_external(f"result_{i}", compressed)

        # === 第三步：汇总生成报告 ===
        print(f"\n[Synthesize] 汇总生成报告...")
        report = self._synthesize(question, results)

        # === 第四步：Reflexion 反思修正 ===
        print(f"\n[Reflexion] 审查报告质量...")
        report = self.reflexion.check_and_refine(report, question)

        return report

    def _plan(self, question: str) -> list[str]:
        """Planner: 拆解研究任务"""
        response = openai.chat.completions.create(
            model="gpt-4o",
            messages=[{
                "role": "user",
                "content": f"""把以下研究问题拆成 3-5 个子任务，每个子任务可独立执行:

问题: {question}

输出 JSON 数组，每个元素是一个子任务描述。"""
            }],
        )
        import json
        try:
            return json.loads(response.choices[0].message.content)
        except:
            return [question]  # fallback: 不拆分

    def _synthesize(self, question: str, results: list[dict]) -> str:
        """汇总所有子任务结果生成报告"""
        # 从外部存储读取压缩后的结果
        result_texts = []
        for i, r in enumerate(results):
            external = self.context.read_external(f"result_{i}")
            result_texts.append(f"## {r['subtask']}\n{external}")

        response = openai.chat.completions.create(
            model="gpt-4o",
            messages=[{
                "role": "user",
                "content": f"""基于以下子任务结果，生成一份结构化研究报告:

研究问题: {question}

子任务结果:
{chr(10).join(result_texts)}

要求: 数据有来源支撑，结论有逻辑链条。"""
            }],
        )
        return response.choices[0].message.content


# === 运行 ===
if __name__ == "__main__":
    agent = ResearchAgent()
    report = agent.run("对比 React 和 Vue 在 2026 年的性能表现")
    print(f"\n{'='*60}")
    print("最终报告:")
    print(report)
```

### 5.2 技术亮点（2分钟）

"架构是分层的：最上层 Plan-and-Execute 拆任务，每个子任务跑 ReAct 循环。关键设计有几个：

第一，工具失败处理。搜索 API 可能超时或限流，我实现了指数退避重试加熔断器——连续失败 3 次自动跳过该工具并降级到备用方案。工具返回的错误信息会做翻译，把技术报错转成 Agent 能理解的提示。

第二，上下文管理。每个子任务用独立上下文窗口避免互相干扰——这就是 Isolate。中间结果 Write 到外部存储不占 token，需要时再 Select 检索。scratchpad 超过 4000 token 自动触发压缩。整个 128K 窗口按比例分配预算。

第三，Reflexion 反思。报告生成后做自我审查，如果发现数据缺来源或逻辑跳跃，生成具体反思存入情景记忆，修改后重新检查，最多重试 2 次。评估优先用外部工具而不是模型自评。"

### 5.3 权衡思考（1分钟）

"为什么用 Plan-and-Execute 而不是纯 ReAct？因为研究任务通常 5 步以上，纯 ReAct 的上下文会爆。为什么子任务用独立上下文？信息隔离避免互相干扰。为什么反思用外部验证而不是自评？纯自评有确认偏差，2024 年研究表明模型自评可能越改越差。"
